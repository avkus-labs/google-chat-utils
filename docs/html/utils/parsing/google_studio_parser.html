 17def load_json_file(file_path: str) -> Optional[Dict[str, Any]]:
 18    """
 19    Reads a JSON-file from the specified path.

 48    def _parse_schema_data(self, schema_dict: Dict[str, Any]) -> SchemaData:
 49        """
 50        Converts a JSON schema dictionary (from Google AI Studio export, assumed to be Gemini-like)
 51        into a SchemaData object.
 52        Google AI Studio's function parameter schemas and response schemas are already
 53        in a format that is quite close to Gemini API's schema structure.
 54        Types are typically uppercase (e.g., "STRING", "OBJECT").
 55        """
 56        if not isinstance(schema_dict, dict):
 57            # This is an error condition, schema_dict should always be a dictionary.
 58            # print(f"Warning: _parse_schema_data expected a dict, got {type(schema_dict)}. Returning default schema.")
 59            return SchemaData(type="STRING", description="Error: Invalid schema format.")
 60
 61        # Directly map Google AI Studio schema to SchemaData fields.
 62        # Ensure type is uppercase as per SchemaData expectation if source varies.
 63        schema_type = str(schema_dict.get("type", "STRING")).upper() # Default to STRING
 64
 65        properties = None
 66        if "properties" in schema_dict and isinstance(schema_dict["properties"], dict):
 67            properties = {
 68                key: self._parse_schema_data(value)
 69                for key, value in schema_dict["properties"].items()
 70            }
 71
 72        items_schema = None
 73        if "items" in schema_dict and isinstance(schema_dict["items"], dict):
 74            items_schema = self._parse_schema_data(schema_dict["items"])
 75
 76        return SchemaData(
 77            type=schema_type,
 78            description=schema_dict.get("description"),
 79            nullable=schema_dict.get("nullable", False), # Default nullable to False
 80            properties=properties,
 81            items=items_schema,
 82            enum=schema_dict.get("enum"),
 83            required=schema_dict.get("required")
 84        )

 86    def _parse_function_definition(self, func_decl_dict: Dict[str, Any]) -> FunctionDefinition:
 87        """
 88        Parses a function declaration dictionary into a FunctionDefinition object.
 89        """
 90        name = func_decl_dict.get("name", "")
 91        description = func_decl_dict.get("description", "")
 92
 93        # 'parameters' in Google AI Studio's export holds the JSON schema for parameters
 94        parameters_json_schema = func_decl_dict.get("parameters", {"type": "OBJECT", "properties": {}}) # Default to empty object schema
 95
 96        # Ensure parameters_json_schema is a dict before parsing
 97        if not isinstance(parameters_json_schema, dict):
 98            # print(f"Warning: 'parameters' for function {name} is not a dict. Using default empty schema.")
 99            parameters_json_schema = {"type": "OBJECT", "properties": {}}
100
101        parameters_schema_data = self._parse_schema_data(parameters_json_schema)
102
103        return FunctionDefinition(
104            name=name,
105            description=description,
106            parameters_schema=parameters_schema_data
107        )

179    def _parse_chat_message_part(self, part_data: Dict[str, Any], owner_role: str = "unknown", is_thought_chunk: bool = False) -> Optional[ChatMessagePart]:
180        """
181        Parses a single part from a chunk into a ChatMessagePart object.
182        Adapted from parser_v2.py's parse_part.
183        The 'owner_role' and 'is_thought_chunk' are contextual hints from the parent chunk.
184        """
185        if not isinstance(part_data, dict):
186            return None
187
188        if 'text' in part_data:
189            # Context (role, isThought) determines if it's TextPart, ThoughtsPart, or (model's) TextPart.
190            # This method creates a generic TextPart; _parse_dialogue_history will refine it.
191            # Or, we can pass context here. Let's pass context.
192            if owner_role == "model" and is_thought_chunk:
193                 return ThoughtsPart(content=part_data['text'])
194            # In parser_v2, 'text_response' was a type applied later. Here, TextPart can cover model's textual response.
195            return TextPart(content=part_data['text'])
196
197        # Multimodal inputs
198        if 'driveImage' in part_data and isinstance(part_data['driveImage'], dict):
199            return UriReferencePart(source_type="google_drive_image", identifier=part_data['driveImage'].get('id'))
200        if 'driveDocument' in part_data and isinstance(part_data['driveDocument'], dict):
201            return UriReferencePart(source_type="google_drive_document", identifier=part_data['driveDocument'].get('id'))
202        if 'sampleImageId' in part_data:
203            return UriReferencePart(source_type="sample_image", identifier=part_data['sampleImageId'])
204        if 'youtubeVideo' in part_data and isinstance(part_data['youtubeVideo'], dict):
205            return UriReferencePart(source_type="youtube_video", identifier=part_data['youtubeVideo'].get('id'))
206        if 'driveVideo' in part_data and isinstance(part_data['driveVideo'], dict):
207            return UriReferencePart(source_type="google_drive_video", identifier=part_data['driveVideo'].get('id'))
208
209        if 'inlineData' in part_data and isinstance(part_data['inlineData'], dict):
210            inline_data = part_data['inlineData']
211            return InlineDataPart(mime_type=inline_data.get('mimeType'), base64_data=inline_data.get('data'))
212
213        if 'fileData' in part_data and isinstance(part_data['fileData'], dict):
214            file_data = part_data['fileData']
215            if 'fileUri' in file_data:
216                 return UriReferencePart(
217                     source_type="file_uri",
218                     identifier=file_data['fileUri'],
219                     mime_type=file_data.get('mimeType')
220                 )
221
222        # Model response parts (tool related)
223        if 'executableCode' in part_data and isinstance(part_data['executableCode'], dict):
224            ec = part_data['executableCode']
225            # Language is not directly in 'executableCode' part in Studio JSON, assume PYTHON or allow override
226            return ExecutableCodePart(code_content=ec.get('code', ''), language=ec.get('language', 'PYTHON'))
227
228        if 'codeExecutionResult' in part_data and isinstance(part_data['codeExecutionResult'], dict):
229            cer = part_data['codeExecutionResult']
230            return CodeExecutionResultPart(outcome=cer.get('outcome', 'OUTCOME_OK'), output_content=cer.get('output', ''))
231
232        if 'functionCall' in part_data and isinstance(part_data['functionCall'], dict):
233            fc = part_data['functionCall']
234            return FunctionCallPart(name=fc.get('name'), args=fc.get('args', {}))
235
236        if 'functionResponse' in part_data and isinstance(part_data['functionResponse'], dict):
237            fr_part = part_data['functionResponse']
238            # 'response' field in a standalone functionResponse part (not from resolvedFunctionCall)
239            # is typically already a dictionary/object.
240            return FunctionResponsePart(name=fr_part.get('name'), response_content=fr_part.get('response', {}))
241
242        # print(f"Warning: Unknown part structure encountered in _parse_chat_message_part: {part_data}")
243        return None

245    def _parse_dialogue_history(self, raw_data: Dict[str, Any]) -> List[DialogueTurn]:
246        """
247        Extracts and transforms dialogue history into a list of DialogueTurn objects.
248        Adapted from extract_dialogue_history_v2 in parser_v2.py.
249        """
250        processed_history: List[DialogueTurn] = []
251
252        source_chunks = raw_data.get('chunkedPrompt', {}).get('chunks', [])
253        if not source_chunks and 'pendingInputs' in raw_data: # Check pendingInputs as alternative source
254            source_chunks = raw_data.get('pendingInputs', [])
255            if not isinstance(source_chunks, list): # Ensure it's a list
256                source_chunks = []
257
258        current_model_turn_parts: List[ChatMessagePart] = []
259        current_model_turn_meta: Dict[str, Any] = {}
260        turn_counter = 1  # Для уникальных turn_id
261
262        def finalize_model_turn():
263            nonlocal current_model_turn_parts, current_model_turn_meta, turn_counter
264            if current_model_turn_parts:
265                turn_to_add = DialogueTurn(
266                    role="model",
267                    parts=list(current_model_turn_parts), # Take a copy
268                    finish_reason=current_model_turn_meta.get('finish_reason'),
269                    token_count=current_model_turn_meta.get('token_count'),
270                    citations=current_model_turn_meta.get('citations'),
271                    turn_id=turn_counter
272                )
273                processed_history.append(turn_to_add)
274                current_model_turn_parts.clear()
275                current_model_turn_meta.clear()
276                turn_counter += 1
277
278        for chunk_index, chunk in enumerate(source_chunks):
279            if not isinstance(chunk, dict):
280                continue
281
282            role = chunk.get('role')
283            is_thought_chunk = chunk.get('isThought', False)
284            chunk_token_count = chunk.get('tokenCount')
285            chunk_citations = chunk.get('citations') # List of dicts
286            chunk_finish_reason = chunk.get('finishReason')
287
288            current_chunk_extracted_parts: List[ChatMessagePart] = []
289
290            # Handle resolvedFunctionCall first (contains both call and response)
291            if role == "model" and 'resolvedFunctionCall' in chunk and isinstance(chunk['resolvedFunctionCall'], dict):
292                rfc = chunk['resolvedFunctionCall']
293                fc_data_from_rfc = rfc.get('functionCall')
294                fr_json_string = rfc.get('response')
295
296                if isinstance(fc_data_from_rfc, dict) and 'name' in fc_data_from_rfc:
297                    # Attempt to remove a pre-existing FunctionCallPart from current_model_turn_parts
298                    # if this resolvedFunctionCall corresponds to it, to avoid duplication.
299                    call_name_from_rfc = fc_data_from_rfc.get('name')
300                    for i, part_in_model_turn in reversed(list(enumerate(current_model_turn_parts))):
301                        if isinstance(part_in_model_turn, FunctionCallPart) and part_in_model_turn.name == call_name_from_rfc:
302                            # To be more precise, one might also compare args if available and simple enough.
303                            # For now, matching by name is the primary heuristic used in parser_v2.
304                            current_model_turn_parts.pop(i)
305                            break
306
307                    # Add the FunctionCall from resolvedFunctionCall itself
308                    current_chunk_extracted_parts.append(FunctionCallPart(name=call_name_from_rfc, args=fc_data_from_rfc.get('args', {})))
309
310                    response_content_dict = {}
311                    if isinstance(fr_json_string, str):
312                        try:
313                            response_content_dict = json.loads(fr_json_string)
314                        except json.JSONDecodeError:
315                            response_content_dict = {"error": "Failed to parse response JSON", "raw_response": fr_json_string}
316                    else:
317                         response_content_dict = {"error": "Response was not a JSON string", "raw_response": fr_json_string}
318
319                    current_chunk_extracted_parts.append(FunctionResponsePart(name=call_name_from_rfc, response_content=response_content_dict))
320
321            # Process parts from the 'parts' array of the chunk
322            # Ensure not to re-process a functionCall if it was part of resolvedFunctionCall handled above for THIS chunk.
323            # However, the main duplication is from previous chunks being merged.
324            chunk_parts_data_raw = chunk.get('parts', [])
325            list_of_part_dicts = []
326            if isinstance(chunk_parts_data_raw, list):
327                list_of_part_dicts = chunk_parts_data_raw
328            elif isinstance(chunk_parts_data_raw, dict): # Single part dict
329                list_of_part_dicts = [chunk_parts_data_raw]
330            elif 'text' in chunk and not chunk_parts_data_raw: # Simple text directly in chunk
331                 list_of_part_dicts = [{'text': chunk['text']}]
332
333            for part_dict_from_chunk in list_of_part_dicts:
334                if not isinstance(part_dict_from_chunk, dict):
335                    continue
336
337                # Pass role and is_thought_chunk for context to _parse_chat_message_part
338                parsed_part = self._parse_chat_message_part(part_dict_from_chunk, owner_role=role, is_thought_chunk=is_thought_chunk)
339                if parsed_part:
340                    current_chunk_extracted_parts.append(parsed_part)
341
342            # Fallback if no parts extracted but chunk has top-level 'text'
343            if not current_chunk_extracted_parts and 'text' in chunk and not list_of_part_dicts:
344                # Role context is important here too
345                text_part_fallback = self._parse_chat_message_part({'text': chunk['text']}, owner_role=role, is_thought_chunk=is_thought_chunk)
346                if text_part_fallback:
347                    current_chunk_extracted_parts.append(text_part_fallback)
348
349            # Refine TextPart for model to ThoughtsPart if it's a thought chunk
350            # This was partially handled in _parse_chat_message_part, but double-check or make logic exclusive.
351            # The current _parse_chat_message_part already creates ThoughtsPart if context is right.
352
353            if role == "user":
354                finalize_model_turn() # Finalize any pending model turn before a user turn
355                if current_chunk_extracted_parts:
356                    user_turn = DialogueTurn(
357                        role="user",
358                        parts=current_chunk_extracted_parts,
359                        token_count=chunk_token_count,
360                        citations=chunk_citations,
361                        turn_id=turn_counter
362                    )
363                    processed_history.append(user_turn)
364            elif role == "model":
365                current_model_turn_parts.extend(current_chunk_extracted_parts)
366                if chunk_finish_reason:
367                    current_model_turn_meta['finish_reason'] = chunk_finish_reason
368                if chunk_token_count: # Accumulate or update token count for the turn
369                    current_model_turn_meta['token_count'] = current_model_turn_meta.get('token_count', 0) + chunk_token_count
370                if chunk_citations:
371                    if 'citations' not in current_model_turn_meta: current_model_turn_meta['citations'] = []
372                    current_model_turn_meta['citations'].extend(chunk_citations if isinstance(chunk_citations, list) else [chunk_citations])
373
374                # Determine if model turn should be finalized
375                is_last_chunk = (chunk_index + 1 == len(source_chunks))
376                next_chunk_is_user = False
377                if not is_last_chunk:
378                    next_chunk = source_chunks[chunk_index + 1]
379                    if isinstance(next_chunk, dict) and next_chunk.get('role') == 'user':
380                        next_chunk_is_user = True
381
382                if chunk_finish_reason or is_last_chunk or next_chunk_is_user:
383                    finalize_model_turn()
384            else: # Unknown or other roles
385                finalize_model_turn()
386                if current_chunk_extracted_parts:
387                    # print(f"Warning: Chunk with role '{role}' found. Storing as is.")
388                    processed_history.append(DialogueTurn(role=role or "unknown", parts=current_chunk_extracted_parts, turn_id=turn_counter))
389
390        finalize_model_turn() # Finalize any last pending model turn
391        print(f'DEBUG _parse_dialogue_history: {processed_history}')
392
393        # Присваиваем turn_id всем ходам, если не был установлен
394        for idx, turn in enumerate(processed_history, 1):
395            if not getattr(turn, 'turn_id', None):
396                turn.turn_id = idx
397        return processed_history

399    def to_markdown(self, intermediate_data: IntermediateChatData) -> str:
400        """
401        Generates a Markdown string with custom XML-like tags from IntermediateChatData.
402        Adapted from format_markdown_for_rag in parser_v2.py.
403        """
404        markdown_parts = []
405        md_add = markdown_parts.append
406
407        md_add(f"# Chat Parse: {intermediate_data.source_file_name}")
408        md_add(f"Parsed at: {intermediate_data.parsed_at_timestamp}")
409        md_add(f"Source Platform: {intermediate_data.source_platform}\n")
410
411        # Generation Settings from RunSettingsData
412        rs = intermediate_data.run_settings
413        md_add("<generation_settings>")
414        if rs.model: md_add(f"  <model>{rs.model}</model>")
415        if rs.temperature is not None: md_add(f"  <temperature>{rs.temperature}</temperature>")
416        if rs.top_p is not None: md_add(f"  <topP>{rs.top_p}</topP>") # Note: key is topP in XML
417        if rs.top_k is not None: md_add(f"  <topK>{rs.top_k}</topK>")   # Note: key is topK in XML
418        if rs.max_output_tokens is not None: md_add(f"  <maxOutputTokens>{rs.max_output_tokens}</maxOutputTokens>")
419        if rs.response_mime_type: md_add(f"  <responseMimeType>{rs.response_mime_type}</responseMimeType>")
420
421        if rs.response_schema:
422            # Assuming SchemaData doesn't have a direct to_json_string method,
423            # we might need SchemaTransformer or a local helper to serialize it for display if needed.
424            # The original parser_v2 used json.dumps on the schema dict.
425            # For now, let's represent its presence or a summary.
426            # If direct JSON representation is needed, SchemaData would need a method,
427            # or we'd use SchemaTransformer here (gemini_schema_to_openai_dict then json.dumps).
428            # For simplicity, using __repr__ or a summary:
429            schema_for_md = intermediate_data.run_settings.raw_settings.get('responseSchema', {})
430            md_add(f"  <response_schema_definition name='configuredResponseSchema'>")
431            md_add("    ```json")
432            md_add(json.dumps(schema_for_md, indent=2, ensure_ascii=False))
433            md_add("    ```")
434            md_add(f"  </response_schema_definition>")
435        md_add("</generation_settings>\n")
436
437        # Configured Tools from ToolConfiguration list
438        if intermediate_data.tool_configurations:
439            md_add("<configured_tools>")
440            for tool_config in intermediate_data.tool_configurations:
441                if tool_config.type == "code_execution":
442                    md_add("  <tool type='code_execution_enabled' />")
443                elif tool_config.type == "google_search_enabled":
444                    md_add("  <tool type='google_search_enabled' />")
445                elif tool_config.type == "browse_enabled":
446                    md_add("  <tool type='browse_enabled' />")
447                elif tool_config.type == "function_declaration" and tool_config.functions:
448                    md_add("  <tool type='function_declarations'>")
449                    for func_def in tool_config.functions:
450                        # Use raw_settings for original schema representation for now
451                        original_func_decl = next((item for item in rs.raw_settings.get('functionDeclarations', []) if item["name"] == func_def.name), None)
452                        params_schema_for_md = {}
453                        if original_func_decl and 'parameters' in original_func_decl:
454                            params_schema_for_md = original_func_decl['parameters']
455
456                        md_add(f"    <function name='{func_def.name}'>")
457                        md_add(f"      <description><![CDATA[{func_def.description}]]></description>")
458                        md_add(f"      <parameters_schema name='{func_def.name}_parameters'>")
459                        md_add("        ```json")
460                        md_add(json.dumps(params_schema_for_md, indent=2, ensure_ascii=False))
461                        md_add("        ```")
462                        md_add(f"      </parameters_schema>")
463                        md_add(f"    </function>")
464                    md_add("  </tool>")
465            md_add("</configured_tools>\n")
466        else:
467            md_add("<tools_placeholder>В этом чате не было активных инструментов (function calling, code execution, search, browse). Для подробностей см. исходный код чата или JSON-представление.</tools_placeholder>\n")
468
469        # System Instruction
470        md_add("<system_instruction>")
471        if intermediate_data.system_instruction and intermediate_data.system_instruction.text.strip():
472            md_add(intermediate_data.system_instruction.text.strip())
473        else:
474            md_add("<!-- No system instruction provided -->")
475        md_add("</system_instruction>")
476
477        # Dialogue History
478        md_add("<dialogue_history>")
479        if not intermediate_data.dialogue_history:
480            md_add("  <!-- No dialogue history available -->")
481        else:
482            for turn_idx, turn in enumerate(intermediate_data.dialogue_history):
483                turn_id_val = turn.turn_id if turn.turn_id else turn_idx+1
484                md_add(f"  <turn role='{turn.role}' id='{turn_id_val}'>")
485                if not turn.parts:
486                    placeholder = "<!-- Empty turn -->"
487                    if turn.role == "user": placeholder = f'    <message_text><!-- Empty user message --></message_text>'
488                    elif turn.role == "model": placeholder = f'    <response_text><!-- Empty model response --></response_text>'
