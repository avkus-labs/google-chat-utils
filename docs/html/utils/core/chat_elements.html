336    @staticmethod
337    def from_dict(data: dict):
338        from utils.core.schema_elements import FunctionDefinition, SchemaData
339        def part_from_dict(part_dict):
340            t = part_dict.get('type')
341            if t == 'TextPart':
342                return TextPart(content=part_dict['content'])
343            if t == 'ThoughtsPart':
344                return ThoughtsPart(content=part_dict['content'])
345            if t == 'ExecutableCodePart':
346                return ExecutableCodePart(code_content=part_dict['code_content'], language=part_dict.get('language', 'PYTHON'))
347            if t == 'CodeExecutionResultPart':
348                return CodeExecutionResultPart(output_content=part_dict['output_content'], outcome=part_dict.get('outcome', 'OK'))
349            if t == 'FunctionCallPart':
350                return FunctionCallPart(name=part_dict['name'], args=part_dict['args'])
351            if t == 'FunctionResponsePart':
352                return FunctionResponsePart(name=part_dict['name'], response_content=part_dict['response_content'])
353            if t == 'InlineDataPart':
354                return InlineDataPart(mime_type=part_dict['mime_type'], base64_data=part_dict['base64_data'])
355            if t == 'UriReferencePart':
356                return UriReferencePart(source_type=part_dict['source_type'], identifier=part_dict['identifier'], mime_type=part_dict.get('mime_type'))
357            return None
358        def turn_from_dict(turn_dict):
359            return DialogueTurn(
360                role=turn_dict['role'],
361                parts=[part_from_dict(p) for p in turn_dict['parts']],
362                turn_id=turn_dict.get('turn_id'),
363                token_count=turn_dict.get('token_count'),
364                finish_reason=turn_dict.get('finish_reason'),
365                citations=turn_dict.get('citations')
366            )
367        def tool_config_from_dict(tc_dict):
368            if tc_dict['type'] == 'function_declaration':
369                return ToolConfiguration(
370                    type='function_declaration',
371                    functions=[FunctionDefinition.from_dict(f) for f in tc_dict.get('functions', [])]
372                )
373            return ToolConfiguration(type=tc_dict['type'])
374        def system_instruction_from_dict(si):
375            if si is None:
376                return None
377            if isinstance(si, dict):
378                return SystemInstruction(text=si.get('text', ''))
379            return SystemInstruction(text=si)
380        return IntermediateChatData(
381            source_file_name=data['source_file_name'],
382            parsed_at_timestamp=data['parsed_at_timestamp'],
383            run_settings=RunSettingsData.from_dict(data['run_settings']),
384            dialogue_history=[turn_from_dict(t) for t in data.get('dialogue_history', [])],
385            source_platform=data.get('source_platform', 'Google AI Studio'),
386            system_instruction=system_instruction_from_dict(data.get('system_instruction')),
